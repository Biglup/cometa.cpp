cmake_minimum_required(VERSION 3.22)
include(cmake/Version.cmake)
project(cometa VERSION ${COMETA_VERSION} LANGUAGES C CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(COMETA_BUILD_TESTS "Build unit tests" ON)
option(COMETA_BUILD_EXAMPLES "Build examples" ON)
option(COMETA_BUILD_DOCS "Build documentation" OFF)
set(COMETA_CARDANO_SOURCE_DIR "" CACHE PATH "Local cardano-c checkout with deep-clone APIs")
include(cmake/Cardano.cmake)

add_library(cometa STATIC lib/src/error.cpp lib/src/blake2b_hash.cpp lib/src/asset_name.cpp lib/src/credential.cpp)
add_library(Cometa::cometa ALIAS cometa)
target_compile_features(cometa PUBLIC cxx_std_17)
target_include_directories(cometa PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/lib/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(cometa PRIVATE "${cardano_source}/lib/include")
target_link_libraries(cometa PRIVATE $<BUILD_INTERFACE:cometa_cardano> $<INSTALL_INTERFACE:Cometa::cardano>)
set_target_properties(cometa PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(cometa PRIVATE /W4 /WX)
else()
  target_compile_options(cometa PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(COMETA_BUILD_TESTS)
  enable_testing()
  find_package(GTest CONFIG REQUIRED)
  add_executable(cometa-tests lib/tests/value_types.cpp)
  target_include_directories(cometa-tests PRIVATE "${cardano_source}/lib" "${cardano_source}/lib/include")
  target_link_libraries(cometa-tests PRIVATE Cometa::cometa GTest::gtest_main)
  add_test(NAME value-types COMMAND cometa-tests)
endif()
if(COMETA_BUILD_EXAMPLES)
  add_executable(cometa-value-types examples/value_types.cpp)
  target_link_libraries(cometa-value-types PRIVATE Cometa::cometa)
endif()
if(COMETA_BUILD_DOCS)
  add_subdirectory(doc)
endif()

install(TARGETS cometa EXPORT CometaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY lib/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${cardano_archive}" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cometa RENAME "${CMAKE_STATIC_LIBRARY_PREFIX}cardano-c${CMAKE_STATIC_LIBRARY_SUFFIX}")
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/cometa)
install(FILES "${cardano_source}/COPYING" DESTINATION ${CMAKE_INSTALL_DATADIR}/cometa RENAME cardano-c-LICENSE)
install(DIRECTORY "${cardano_source}/lib/external/" DESTINATION ${CMAKE_INSTALL_DATADIR}/cometa/third-party FILES_MATCHING PATTERN "*LICENSE*" PATTERN "*COPYING*" PATTERN "*NOTICE*" PATTERN "*license*" PATTERN "*copyright*")
install(EXPORT CometaTargets NAMESPACE Cometa:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Cometa)
configure_package_config_file(cmake/CometaConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/CometaConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Cometa)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CometaConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CometaConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/CometaConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Cometa)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_VERSION "${COMETA_FULL_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "cometa-${CPACK_PACKAGE_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}-${CMAKE_CXX_COMPILER_ID}")
include(CPack)
